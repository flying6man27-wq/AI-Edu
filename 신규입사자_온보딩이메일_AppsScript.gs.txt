function _send(list, label) {
  var sent = 0, failed = 0;
  for (var i = 0; i < list.length; i++) {
    var r = list[i];
    try {
      GmailApp.sendEmail(r.email, SUBJECT, buildBody(r.name, r.org, r.joined));
      sent++;
      Logger.log('OK [' + (i+1) + '/' + list.length + '] ' + r.name + ' (' + r.org + ') <' + r.email + '>');
      Utilities.sleep(400);
    } catch(e) {
      Logger.log('FAIL ' + r.name + ' <' + r.email + '> : ' + e.message);
      failed++;
    }
  }
  Logger.log('[' + label + '] 완료 — 발송 ' + sent + '건 / 실패 ' + failed + '건');
}

function _drafts(list, label) {
  for (var i = 0; i < list.length; i++) {
    var r = list[i];
    GmailApp.createDraft(r.email, SUBJECT, buildBody(r.name, r.org, r.joined));
    Logger.log('초안 ' + r.name + ' (' + r.org + ') <' + r.email + '>');
  }
  Logger.log('[' + label + '] 초안 ' + list.length + '건 생성 — Gmail 임시보관함 확인');
}
