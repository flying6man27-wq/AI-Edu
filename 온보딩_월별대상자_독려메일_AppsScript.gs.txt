// ───────────────────────── 월 추출 유틸 ───────────────────────────
function joinedMonth_(joined) {
  var m = String(joined).match(/(\d+)\s*월/);
  return m ? parseInt(m[1], 10) : 0;
}
// TARGET_MONTHS 에 해당하는 대상자만 반환
function targetsByMonth_(months) {
  months = months || TARGET_MONTHS;
  return ROSTER.filter(function(r){
    return months.indexOf(joinedMonth_(r.joined)) >= 0;
  });
}

// ───────────── Firestore REST 헬퍼 (운영에이전트와 동일 방식) ─────────────
function fsBase_(){
  return 'https://firestore.googleapis.com/v1/projects/'+PROJECT_ID+
         '/databases/(default)/documents';
}
function fsHeaders_(){ return { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() }; }

function decodeValue_(v){
  if(v==null) return null;
  if('stringValue'  in v) return v.stringValue;
  if('integerValue' in v) return parseInt(v.integerValue,10);
  if('doubleValue'  in v) return v.doubleValue;
  if('booleanValue' in v) return v.booleanValue;
  if('timestampValue' in v) return v.timestampValue;
  if('mapValue'   in v) return decodeFields_(v.mapValue.fields || {});
  if('arrayValue' in v) return (v.arrayValue.values||[]).map(decodeValue_);
  if('nullValue'  in v) return null;
  return null;
}
function decodeFields_(fields){
  var o = {};
  Object.keys(fields).forEach(function(k){ o[k] = decodeValue_(fields[k]); });
  return o;
}

// 컬렉션 전체를 페이지네이션으로 읽기
function fsList_(collection){
  var out = [], pageToken = '';
  do {
    var url = fsBase_() + '/' + collection + '?pageSize=300' +
              (pageToken ? '&pageToken='+encodeURIComponent(pageToken) : '');
    var res = UrlFetchApp.fetch(url, { headers: fsHeaders_(), muteHttpExceptions: true });
    if(res.getResponseCode() !== 200){
      throw new Error('Firestore 읽기 실패 ['+collection+'] ('+res.getResponseCode()+'): '+res.getContentText());
    }
    var body = JSON.parse(res.getContentText());
    (body.documents || []).forEach(function(doc){
      var o = decodeFields_(doc.fields || {});
      o.__id = doc.name.split('/').pop();
      out.push(o);
    });
    pageToken = body.nextPageToken || '';
  } while(pageToken);
  return out;
}

/**
 * ★ 최초 1회 실행. 온보딩 이수현황이 어느 컬렉션에 있는지 탐색합니다.
 *   대시보드 온보딩 탭 코드를 확보하지 못해 컬렉션명이 확정되지 않은 상태입니다.
 *   로그에서 문서가 잡히는 이름을 찾아 OB_COLLECTION 에 반영하세요.
 */
function probeCollections(){
  var candidates = ['onboarding_learners','onboarding','onboardingLearners',
                    'ob_learners','onboarding_progress','learners'];
  candidates.forEach(function(c){
    try {
      var rows = fsList_(c);
      Logger.log('✔ [' + c + '] 문서 ' + rows.length + '건');
      if(rows.length){
        Logger.log('    샘플 필드: ' + JSON.stringify(Object.keys(rows[0])));
        Logger.log('    샘플 값  : ' + JSON.stringify(rows[0]).slice(0,500));
      }
    } catch(e){
      Logger.log('✘ [' + c + '] ' + e.message.split('\n')[0]);
    }
  });
}

// ───────────────────────── 진도 계산 ──────────────────────────────
// 이메일(소문자)로 온보딩 학습자 문서를 찾아 이수 모듈 수를 반환
function buildProgressIndex_(){
  var idx = {};
  var rows;
  try { rows = fsList_(OB_COLLECTION); }
  catch(e){ Logger.log('⚠ 온보딩 컬렉션 읽기 실패 — 진도 없이 진행합니다: '+e.message.split('\n')[0]); return idx; }

  rows.forEach(function(l){
    var key = String(l.email || '').trim().toLowerCase();
    if(!key) return;
    var set = {};
    if(l.modules && typeof l.modules === 'object') for(var k in l.modules) set[k] = l.modules[k];
    for(var kk in l){ var m = kk.match(/^modules\.(.+)$/); if(m) set[m[1]] = l[kk]; }

    var done = [];
    OB_MODULES.forEach(function(id){
      var mod = set[id];
      if(mod && (mod.completed === true || mod === true)) done.push(id);
    });
    // modules 구조가 없고 completedCount 만 있는 경우 대비
    var cnt = done.length || (typeof l.completedCount === 'number' ? l.completedCount : 0);
    idx[key] = { done: done, count: cnt, raw: l };
  });
  return idx;
}

// ───────────────────── 대상자 + 진도 결합 ─────────────────────────
function resolveTargets_(months){
  var targets = targetsByMonth_(months);
  var idx = buildProgressIndex_();
  return targets.map(function(r){
    var prog = idx[String(r.email).trim().toLowerCase()] || null;
    var cnt  = prog ? prog.count : 0;
    return {
      r: r, prog: prog, count: cnt,
      done: (cnt >= OB_MODULES.length),
      started: !!prog
    };
  });
}

/**
 * 드라이런 — 메일을 보내지 않고 대상자와 진도만 로그로 확인합니다.
 * 실제 발송 전 반드시 먼저 실행하세요.
 */
function previewTargets(){
  var list = resolveTargets_();
  Logger.log('대상 입사월: ' + TARGET_MONTHS.join(', ') + '월');
  Logger.log('추출 인원  : ' + list.length + '명 / 전체 명단 ' + ROSTER.length + '명');
  Logger.log('마감일     : ' + DEADLINE);
  Logger.log('─────────────────────────────────────────────');
  if(!list.length){
    Logger.log('※ 해당 월 대상자가 명단에 없습니다. ROSTER 에 해당 월 입사자를 추가하세요.');
    return;
  }
  list.forEach(function(t){
    Logger.log(
      (t.done ? '[이수완료] ' : (t.started ? '[진행중  ] ' : '[미시작  ] ')) +
      joinedMonth_(t.r.joined) + '월  ' + t.r.name + ' (' + t.r.org + ')  ' +
      t.count + '/' + OB_MODULES.length + '  <' + t.r.email + '>'
    );
  });
  var need = list.filter(function(t){ return !t.done; });
  Logger.log('─────────────────────────────────────────────');
  Logger.log('발송 대상(미이수): ' + need.length + '명 / 이수완료 제외: ' + (list.length-need.length) + '명');
}

// ───────────────────────── 발송 함수 ──────────────────────────────
function _run(dryDraft, months){
  var list = resolveTargets_(months);
  var need = list.filter(function(t){ return !t.done; });   // 이수완료자는 제외
  var sent = 0, failed = 0;

  need.forEach(function(t, i){
    // 학습 이력이 있으면 독려메일, 전혀 없으면 신규 안내메일
    var subject = t.started ? SUBJECT_NUDGE : SUBJECT_NEW;
    var body    = t.started ? buildNudgeBody(t.r, t.prog) : buildNewBody(t.r);
    try {
      if(dryDraft) GmailApp.createDraft(t.r.email, subject, body);
      else         GmailApp.sendEmail(t.r.email, subject, body);
      sent++;
      Logger.log((dryDraft?'초안':'발송') + ' [' + (i+1) + '/' + need.length + '] ' +
                 t.r.name + ' (' + t.r.org + ') ' + t.count + '/' + OB_MODULES.length +
                 ' <' + t.r.email + '>');
      if(!dryDraft) Utilities.sleep(400);
    } catch(e){
      Logger.log('FAIL ' + t.r.name + ' <' + t.r.email + '> : ' + e.message);
      failed++;
    }
  });
  Logger.log('완료 — ' + (dryDraft?'초안 ':'발송 ') + sent + '건 / 실패 ' + failed +
             '건 / 이수완료 제외 ' + (list.length - need.length) + '건');
}

function draftNudge(){ _run(true);  }   // 초안만 생성
function sendNudge(){  _run(false); }   // 실제 발송
